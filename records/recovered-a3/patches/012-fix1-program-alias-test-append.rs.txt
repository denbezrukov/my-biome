
#[cfg(unix)]
#[test]
fn inventory_review_tsconfig_program_alias_keeps_program_protection() {
    let mut fixture = biome_fs::TemporaryFs::new("biome-a3-review-program-alias");
    fixture.create_file("package.json", "{}");
    fixture.create_file("tsconfig.json", r#"{"files":["alias.ts"]}"#);
    let target = fixture.create_file("real.ts", "export {};");
    std::os::unix::fs::symlink(target, fixture.working_directory.join("alias.ts")).unwrap();
    let (service, project) = setup_workspace_and_open_project(fixture.create_os(), fixture.cli_path());
    let inventory = service.discover_inventory(project, &ProjectAnalysisRequest { configuration:serde_json::from_value(serde_json::json!({"entry":[],"useTsconfigFiles":true})).unwrap(), ..Default::default() }).unwrap();
    assert_eq!(root_paths(&inventory), ["real.ts".into()].into());
    assert!(inventory.files[0].program);
    assert!(inventory.gaps.is_empty(), "{:?}", inventory.gaps);
}
