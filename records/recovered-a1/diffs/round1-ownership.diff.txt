+#[test]
+fn native_language_fixtures_have_native_owners() {
+    let inventory: serde_json::Value = serde_json::from_str(include_str!(
+        "../../../xtask/project_analysis/reference/tests.json"
+    ))
+    .unwrap();
+    let groups = inventory["groups"].as_array().unwrap();
+    for (fixture, owner) in [
+        ("commonjs-tsconfig", "B2"),
+        ("commonjs", "B2"),
+        ("custom-elements", "B2"),
+        ("js-only", "B1"),
+        ("jsdoc-exports", "B2"),
+        ("jsdoc-import-tag", "B2"),
+        ("jsdoc", "B5"),
+        ("jsx", "B2"),
+        ("tagged-template-literal", "B2"),
+    ] {
+        let path = format!("packages/knip/test/language/{fixture}.test.ts");
+        let group = groups.iter().find(|group| group["path"] == path).unwrap();
+        assert_eq!(group["owner"], owner, "{path}");
+        assert_eq!(group["status"], "untested", "{path}");
+    }
+    let deferred = groups
+        .iter()
+        .find(|group| {
+            group["path"] == "packages/knip/test/language/svelte-script-jsdoc-import.test.ts"
+        })
+        .unwrap();
+    assert_eq!(deferred["owner"], "deferred extensions");
+    assert_eq!(deferred["status"], "unsupported");
+}
